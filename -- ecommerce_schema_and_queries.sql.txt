-- ecommerce_schema_and_queries.sql
-- Create tables (works for MySQL / SQLite with small differences)
PRAGMA foreign_keys = ON; -- SQLite only; harmless in MySQL

-- DROP if exists (so repeated runs are safe)
DROP TABLE IF EXISTS order_items;
DROP TABLE IF EXISTS orders;
DROP TABLE IF EXISTS products;
DROP TABLE IF EXISTS users;

-- USERS
CREATE TABLE users (
  user_id INTEGER PRIMARY KEY,
  user_name TEXT,
  email TEXT,
  created_date DATE
);

-- PRODUCTS
CREATE TABLE products (
  product_id INTEGER PRIMARY KEY,
  product_name TEXT,
  category TEXT,
  price NUMERIC
);

-- ORDERS
CREATE TABLE orders (
  order_id INTEGER PRIMARY KEY,
  user_id INTEGER,
  order_date DATE,
  status TEXT,
  FOREIGN KEY (user_id) REFERENCES users(user_id)
);

-- ORDER ITEMS
CREATE TABLE order_items (
  order_item_id INTEGER PRIMARY KEY,
  order_id INTEGER,
  product_id INTEGER,
  quantity INTEGER,
  item_price NUMERIC, -- price at time of order
  FOREIGN KEY (order_id) REFERENCES orders(order_id),
  FOREIGN KEY (product_id) REFERENCES products(product_id)
);

-- Add indexes for optimization (hint f)
CREATE INDEX IF NOT EXISTS idx_orders_user ON orders(user_id);

-- -------------------------------
-- Example queries (deliverables)
-- -------------------------------

-- 1) Basic SELECT, WHERE, ORDER BY
-- Recent completed orders:
SELECT o.order_id, o.order_date, u.user_name, o.status
FROM orders o
JOIN users u ON o.user_id = u.user_id
WHERE o.status = 'completed'
ORDER BY o.order_date DESC;

-- 2) GROUP BY + aggregate functions
-- Total revenue per order:
SELECT oi.order_id,
       SUM(oi.quantity * oi.item_price) AS order_revenue
FROM order_items oi
GROUP BY oi.order_id
ORDER BY order_revenue DESC;

-- 3) Average revenue per user (two ways)
-- A) Per-user average of their order totals (preferred for per-user metric)
SELECT u.user_id, u.user_name,
       AVG(ord.revenue) AS avg_revenue_per_user
FROM users u
JOIN (
  SELECT o.user_id, o.order_id, SUM(oi.quantity * oi.item_price) AS revenue
  FROM orders o
  JOIN order_items oi ON o.order_id = oi.order_id
  WHERE o.status = 'completed'
  GROUP BY o.order_id
) ord ON u.user_id = ord.user_id
GROUP BY u.user_id, u.user_name
ORDER BY avg_revenue_per_user DESC;

-- B) Global average revenue per user: total revenue / number of unique users
SELECT
  (SUM(oi.quantity * oi.item_price) * 1.0) / COUNT(DISTINCT o.user_id) AS avg_revenue_per_user_global
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
WHERE o.status = 'completed';

-- 4) JOINS examples (INNER, LEFT)
-- INNER JOIN: products sold with quantities
SELECT p.product_name, SUM(oi.quantity) AS total_qty
FROM products p
JOIN order_items oi ON p.product_id = oi.product_id
GROUP BY p.product_id, p.product_name
ORDER BY total_qty DESC;

-- LEFT JOIN: list all users and their order count (including zero)
SELECT u.user_id, u.user_name, COUNT(o.order_id) AS orders_count
FROM users u
LEFT JOIN orders o ON u.user_id = o.user_id
GROUP BY u.user_id, u.user_name;

-- 5) Subquery example
-- Users who spent more than avg user spend
WITH per_user AS (
  SELECT o.user_id, SUM(oi.quantity * oi.item_price) AS total_spend
  FROM orders o
  JOIN order_items oi ON o.order_id = oi.order_id
  WHERE o.status = 'completed'
  GROUP BY o.user_id
)
SELECT p.user_id, u.user_name, p.total_spend
FROM per_user p
JOIN users u ON p.user_id = u.user_id
WHERE p.total_spend > (SELECT AVG(total_spend) FROM per_user);

-- 6) Create a view for analysis (f)
CREATE VIEW IF NOT EXISTS vw_order_revenue AS
SELECT o.order_id, o.user_id, o.order_date, o.status,
       SUM(oi.quantity * oi.item_price) AS revenue
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
GROUP BY o.order_id;

-- Select from view
SELECT * FROM vw_order_revenue ORDER BY revenue DESC;

-- 7) Handling NULL values (example)
-- Suppose some item_price is NULL, we treat as 0
SELECT oi.order_id, SUM(oi.quantity * COALESCE(oi.item_price,0)) AS revenue_handling_nulls
FROM order_items oi
GROUP BY oi.order_id;

-- 8) Optimizing tips shown via indexes earlier; example of using an indexed query
EXPLAIN QUERY PLAN
SELECT o.order_id
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
WHERE oi.product_id = 2; -- index on order_items.product_id helps
